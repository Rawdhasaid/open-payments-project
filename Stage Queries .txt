CREATE SCHEMA IF NOT EXISTS stg;

DROP TABLE IF EXISTS stg.general_payments_2025_clean;

CREATE TABLE stg.general_payments_2025_clean AS
TABLE raw.general_payments_2025;

UPDATE stg.general_payments_2025_clean
SET
    "Covered_Recipient_Name_Suffix" =
        NULLIF(UPPER(TRIM("Covered_Recipient_Name_Suffix")), ''),

    "Covered_Recipient_First_Name" =
        NULLIF(UPPER(TRIM("Covered_Recipient_First_Name")), ''),

    "Covered_Recipient_Middle_Name" =
        NULLIF(UPPER(TRIM("Covered_Recipient_Middle_Name")), ''),

    "Covered_Recipient_Last_Name" =
        NULLIF(UPPER(TRIM("Covered_Recipient_Last_Name")), ''),

    "Recipient_City" =
        NULLIF(UPPER(TRIM("Recipient_City")), ''),

    "Recipient_Country" =
        NULLIF(UPPER(TRIM("Recipient_Country")), ''),

    "Recipient_Province" =
        NULLIF(UPPER(TRIM("Recipient_Province")), ''),

    "Recipient_State" =
        NULLIF(UPPER(TRIM("Recipient_State")), ''),

    "Recipient_Postal_Code" =
        NULLIF(TRIM("Recipient_Postal_Code"), ''),

    "Recipient_Zip_Code" =
        NULLIF(TRIM("Recipient_Zip_Code"), ''),

    "Recipient_Primary_Business_Street_Address_Line1" =
        NULLIF(UPPER(TRIM("Recipient_Primary_Business_Street_Address_Line1")), ''),

    "Recipient_Primary_Business_Street_Address_Line2" =
        NULLIF(UPPER(TRIM("Recipient_Primary_Business_Street_Address_Line2")), ''),

    "Covered_Recipient_Specialty_1" =
        NULLIF(UPPER(TRIM("Covered_Recipient_Specialty_1")), ''),

    "Covered_Recipient_Specialty_2" =
        NULLIF(UPPER(TRIM("Covered_Recipient_Specialty_2")), '');
        
        
        SELECT COUNT(*)
FROM stg.general_payments_2025_clean;
